# Runtime dependencies for the robot itself.
#
# Install on the board with:
#     pip install --no-cache-dir -r requirements.txt
#
# --no-cache-dir matters on a 64 GB SD card: pip's wheel cache can otherwise
# consume several hundred megabytes you will want for models.
#
# Piper is NOT installed from PyPI here. On arm64 the prebuilt release binary
# from https://github.com/rhasspy/piper/releases is smaller and faster than the
# Python package and its onnxruntime dependency; scripts/fetch_models.sh
# downloads it. Set tts.binary in config.toml if you install it elsewhere.

tomli==2.0.1 ; python_version < "3.11"
# ^ tomllib landed in the standard library in 3.11. Radxa OS is Debian 11,
# which ships Python 3.9, so the backport is needed there and skipped on any
# newer image. walle/config.py imports whichever one exists.

spidev==3.6                # /dev/spidev access for the ILI9341 panel; only
                           # needed when display.backend resolves to "spi"

vosk==0.3.45              # offline speech recognition (~100 MB with the small model)
PyAudio==0.2.14           # PortAudio bindings for the INMP441 capture stream
argostranslate==1.9.6     # offline translation; language packs installed separately

# The Gemini online backend deliberately uses stdlib urllib rather than requests
# or the Google SDK: it is one HTTP POST, and on a 1 GB board the SDK's
# transitive dependencies cost more RAM than the feature is worth.

# gpiod comes from the distribution, not pip: the Python bindings must match the
# libgpiod version Armbian ships, and the pip package frequently does not.
#     sudo apt install python3-libgpiod gpiod
